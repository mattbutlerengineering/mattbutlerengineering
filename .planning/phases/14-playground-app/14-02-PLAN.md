```typescript
export function relativeTime(date: Date): string {
  const rtf = new Intl.RelativeTimeFormat("en", { numeric: "auto" });
  const diffMs = date.getTime() - Date.now();
  const diffSecs = Math.round(diffMs / 1000);
  const diffMins = Math.round(diffSecs / 60);
  if (Math.abs(diffSecs) < 60) return rtf.format(diffSecs, "second");
  if (Math.abs(diffMins) < 60) return rtf.format(diffMins, "minute");
  const diffHours = Math.round(diffMins / 60);
  return rtf.format(diffHours, "hour");
}
```

**PlaygroundPage.tsx:**

State management:
- `const [history, setHistory] = useState<HistoryEntry[]>([])` — prompt history, capped at 50
- `const [activeId, setActiveId] = useState<string | null>(null)` — currently selected history entry
- `const promptRef = useRef("")` — tracks the prompt that was just submitted (use ref to avoid stale closure in onComplete)

Streaming hook:
```typescript
const rawLinesRef = useRef<string[]>([]);
// Keep rawLinesRef in sync
rawLinesRef.current = rawLines;

const { spec, isStreaming, error, rawLines, send, clear, usage } = useGenStream({
  api: "/api/gen/ui",
  onComplete: (completedSpec) => {
    const entry: HistoryEntry = {
      id: crypto.randomUUID(),
      prompt: promptRef.current,
      spec: completedSpec,
      rawLines: [...rawLinesRef.current],
      timestamp: new Date(),
    };
    setHistory((prev) => [entry, ...prev].slice(0, 50));
    setActiveId(entry.id);
  },
});
```

Display logic — two modes:
1. **Live streaming mode** (isStreaming is true, or no active history entry): show `spec`, `rawLines`, `isStreaming`, `error` from useGenStream directly
2. **History review mode** (activeId matches a completed history entry and not streaming): show `entry.spec`, `entry.rawLines`, `isStreaming=false`, `error=null` from the selected entry

```typescript
const activeEntry = activeId ? history.find(e => e.id === activeId) : null;
const displaySpec = isStreaming ? spec : (activeEntry?.spec ?? spec);
const displayRawLines = isStreaming ? rawLines : (activeEntry?.rawLines ?? rawLines);
const displayError = isStreaming ? error : null;
```

Event handlers:
- `handleSubmit(prompt)`: store prompt in promptRef, call `send(prompt)`, set activeId to null (switch to live mode)
- `handleStop()`: abort the stream (useGenStream handles this via AbortController — the partial spec is kept)
- `handleSelectHistory(id)`: set activeId to selected entry (only when not streaming)
- `handleRetry()`: re-send the active entry's prompt or promptRef.current
