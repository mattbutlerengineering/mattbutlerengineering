```typescript
interface SidebarSection {
  label?: string;
  items: SidebarItem[];
}
interface SidebarItem {
  id: string;
  label: string;
  icon?: ReactNode;
  href?: string;       // DO NOT USE — causes full page reloads

  disabled?: boolean;
  onClick?: () => void; // USE THIS with useNavigate
}
// Usage: <Sidebar items={sections} />
```

```typescript
interface CardProps extends HTMLAttributes<HTMLDivElement> {
  variant?: "elevated" | "glass" | "flat";
  tilt?: boolean;
  title?: string;
  subtitle?: string;
  children?: ReactNode;
}
// Replaces: Card + CardHeader + CardTitle + CardContent from @mbe/ui
// Simple: <Card title="Quick Stats">content</Card>
// Complex header: <Card><div className={styles.cardHeader}>...</div>content</Card>
```
